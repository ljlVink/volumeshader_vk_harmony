cmake_minimum_required(VERSION 3.22)
project(volumeshader_bmvk LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)

# HarmonyOS SDK root that contains openharmony/ and hms/ (possibly below a
# default/ level). Resolved from the HARMONY_SDK environment variable, or from
# HMOS_SDK_NATIVE when DevEco/hvigor injects it; no hardcoded host paths.
set(HARMONY_SDK "$ENV{HARMONY_SDK}" CACHE PATH "Harmony SDK root containing default/ or openharmony/")
if(NOT HARMONY_SDK AND HMOS_SDK_NATIVE)
    get_filename_component(HARMONY_SDK "${HMOS_SDK_NATIVE}/../.." ABSOLUTE)
endif()
if(EXISTS "${HARMONY_SDK}/default/openharmony")
    set(HARMONY_SDK "${HARMONY_SDK}/default")
endif()
find_program(GLSLANG_VALIDATOR NAMES glslang_validator glslangValidator
    HINTS "${HARMONY_SDK}/openharmony/toolchains" "$ENV{VULKAN_SDK}/Bin" REQUIRED)

set(SHADER_BINARY_DIR "${CMAKE_CURRENT_BINARY_DIR}/shaders")
file(MAKE_DIRECTORY "${SHADER_BINARY_DIR}")
set(SHADER_FILES volume.vert volume_reference.frag volume_optimized.frag present.vert present.frag)
set(SHADER_OUTPUTS)
foreach(SHADER IN LISTS SHADER_FILES)
    set(SOURCE "${CMAKE_CURRENT_SOURCE_DIR}/shaders/${SHADER}")
    set(DEFINES)
    if(SHADER MATCHES "^volume_.*\\.frag$")
        set(SOURCE "${CMAKE_CURRENT_SOURCE_DIR}/shaders/volume.frag")
        if(SHADER STREQUAL "volume_optimized.frag")
            set(DEFINES -DOPTIMIZED_KERNEL=1)
        else()
            set(DEFINES -DOPTIMIZED_KERNEL=0)
        endif()
    endif()
    set(SPV "${SHADER_BINARY_DIR}/${SHADER}.spv")
    add_custom_command(OUTPUT "${SPV}"
        COMMAND "${GLSLANG_VALIDATOR}" -V --target-env vulkan1.1 ${DEFINES} -o "${SPV}" "${SOURCE}"
        DEPENDS "${SOURCE}" "${CMAKE_CURRENT_SOURCE_DIR}/shaders/kernel.glsl" VERBATIM)
    list(APPEND SHADER_OUTPUTS "${SPV}")
endforeach()
set(SHADER_HEADER "${CMAKE_CURRENT_BINARY_DIR}/generated/volume_shaders.h")
add_custom_command(OUTPUT "${SHADER_HEADER}"
    COMMAND "${CMAKE_COMMAND}" "-DINPUT_DIR=${SHADER_BINARY_DIR}" "-DOUTPUT=${SHADER_HEADER}"
        -P "${CMAKE_CURRENT_SOURCE_DIR}/cmake/embed_shaders.cmake"
    DEPENDS ${SHADER_OUTPUTS} "${CMAKE_CURRENT_SOURCE_DIR}/cmake/embed_shaders.cmake" VERBATIM)
add_library(volume_core STATIC src/renderer.cpp "${SHADER_HEADER}")
target_include_directories(volume_core PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/include"
    PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/generated")

if(CMAKE_SYSTEM_NAME STREQUAL "OHOS")
    target_compile_definitions(volume_core PUBLIC VK_USE_PLATFORM_OHOS PRIVATE VOLUME_HMS=1)
    if(HMOS_SDK_NATIVE)
        set(HMS_NATIVE "${HMOS_SDK_NATIVE}")
    else()
        set(HMS_NATIVE "${HARMONY_SDK}/hms/native")
    endif()
    target_include_directories(volume_core PRIVATE "${HMS_NATIVE}/sysroot/usr/include")
    if(OHOS_ARCH STREQUAL "x86_64")
        set(HMS_ARCH x86_64-linux-ohos)
    elseif(OHOS_ARCH STREQUAL "armeabi-v7a")
        set(HMS_ARCH arm-linux-ohos)
    else()
        set(HMS_ARCH aarch64-linux-ohos)
    endif()
    find_library(HMS_FG framegeneration PATHS "${HMS_NATIVE}/sysroot/usr/lib/${HMS_ARCH}" NO_DEFAULT_PATH NO_CMAKE_FIND_ROOT_PATH REQUIRED)
    find_library(HMS_OPENGTX opengtx PATHS "${HMS_NATIVE}/sysroot/usr/lib/${HMS_ARCH}" NO_DEFAULT_PATH NO_CMAKE_FIND_ROOT_PATH REQUIRED)
    find_library(HMS_XENGINE xengine PATHS "${HMS_NATIVE}/sysroot/usr/lib/${HMS_ARCH}" NO_DEFAULT_PATH NO_CMAKE_FIND_ROOT_PATH REQUIRED)
    target_link_libraries(volume_core PUBLIC vulkan ${HMS_FG} ${HMS_OPENGTX} ${HMS_XENGINE} dl)
elseif(WIN32)
    include(FetchContent)
    FetchContent_Declare(vulkan_headers URL https://codeload.github.com/KhronosGroup/Vulkan-Headers/zip/refs/tags/v1.3.296 DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
    FetchContent_MakeAvailable(vulkan_headers)
    target_link_libraries(volume_core PUBLIC Vulkan::Headers)
    target_compile_definitions(volume_core PUBLIC VK_USE_PLATFORM_WIN32_KHR NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0601 WINVER=0x0601)
    if(MINGW)
        # Older PE linkers cannot consume recent Windows DLLs directly. Generate an ordinary
        # import library from the authoritative Vulkan prototypes; only referenced entries are linked.
        set(VULKAN_DEF "${CMAKE_CURRENT_BINARY_DIR}/vulkan-loader.def")
        file(STRINGS "${vulkan_headers_SOURCE_DIR}/include/vulkan/vulkan_core.h" VULKAN_PROTOTYPES REGEX "^VKAPI_ATTR")
        set(VULKAN_EXPORTS "LIBRARY vulkan-1.dll\nEXPORTS\n")
        foreach(PROTOTYPE IN LISTS VULKAN_PROTOTYPES)
            if(PROTOTYPE MATCHES "VKAPI_CALL (vk[A-Za-z0-9_]+)\\(")
                string(APPEND VULKAN_EXPORTS "${CMAKE_MATCH_1}\n")
            endif()
        endforeach()
        file(WRITE "${VULKAN_DEF}" "${VULKAN_EXPORTS}")
        set(VULKAN_IMPORT "${CMAKE_CURRENT_BINARY_DIR}/libvulkan-loader.a")
        add_custom_command(OUTPUT "${VULKAN_IMPORT}"
            COMMAND "${CMAKE_DLLTOOL}" --input-def "${VULKAN_DEF}" --dllname vulkan-1.dll --output-lib "${VULKAN_IMPORT}"
            DEPENDS "${VULKAN_DEF}" VERBATIM)
        add_custom_target(volume_vulkan_import DEPENDS "${VULKAN_IMPORT}")
        add_library(volume_vulkan_loader STATIC IMPORTED)
        set_target_properties(volume_vulkan_loader PROPERTIES IMPORTED_LOCATION "${VULKAN_IMPORT}")
        add_dependencies(volume_core volume_vulkan_import)
        target_link_libraries(volume_core PUBLIC volume_vulkan_loader)
    else()
        find_package(Vulkan REQUIRED)
        target_link_libraries(volume_core PUBLIC Vulkan::Vulkan)
    endif()
else()
    message(FATAL_ERROR "Supported hosts: HarmonyOS native and Windows desktop")
endif()
if(CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
    target_compile_options(volume_core PRIVATE -Wall -Wextra -Wpedantic)
endif()
